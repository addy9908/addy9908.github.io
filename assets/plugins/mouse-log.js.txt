my_widget_script =
{
  // ================= data =================
  mw_mice: [],
  mw_events: [],
  mw_editId: null,
  mw_editEventIdx: null,
  mw_stagePhotos: [],          // shared photos for the event being composed
  mw_stagePerMouse: {},        // { mouseId: "note" }
  mw_stagePerMousePhotos: {},  // { mouseId: [dataUrl, ...] }  <-- per-mouse photos
  mw_collapsed: {},
  mw_rosterCollapsed: false,

  // ================= helpers =================
  mw_esc: function (s) {
    if (s === null || s === undefined) { return ''; }
    s = String(s);
    s = s.split('&').join('&amp;');
    s = s.split('<').join('&lt;');
    s = s.split('>').join('&gt;');
    s = s.split('"').join('&quot;');
    return s;
  },

  mw_today: function () {
    var d = new Date();
    var mm = ('0' + (d.getMonth() + 1)).slice(-2);
    var dd = ('0' + d.getDate()).slice(-2);
    return d.getFullYear() + '-' + mm + '-' + dd;
  },

  mw_trim: function (s) {
    if (s === null || s === undefined) { return ''; }
    return String(s).replace(/^\s+/, '').replace(/\s+$/, '');
  },

  mw_sortEvents: function () {
    var self = this;
    var dir = 1;
    var ssel = document.getElementById('sortorder');
    if (ssel && ssel.value === 'newest') { dir = -1; }
    this.mw_events.sort(function (a, b) {
      var da = self.mw_parseDate(a.date);
      var db = self.mw_parseDate(b.date);
      var ta = da ? da.getTime() : 0;
      var tb = db ? db.getTime() : 0;
      if (ta < tb) { return -1 * dir; }
      if (ta > tb) { return 1 * dir; }
      return 0;
    });
  },

  mw_ageAt: function (dob, eventDate) {
    var d0 = this.mw_parseDate(dob);
    var d1 = this.mw_parseDate(eventDate);
    if (!d0 || !d1) { return ''; }
    var days = Math.floor((d1.getTime() - d0.getTime()) / 86400000);
    if (days < 0) { return ''; }
    return days + 'd (' + Math.floor(days / 7) + 'w)';
  },

  mw_parseDate: function (s) {
    if (!s) { return null; }
    s = this.mw_trim(s);
    var m;
    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
    if (m) { return new Date(parseInt(m[3],10), parseInt(m[1],10)-1, parseInt(m[2],10)); }
    m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
    if (m) { return new Date(parseInt(m[1],10), parseInt(m[2],10)-1, parseInt(m[3],10)); }
    var d = new Date(s);
    return isNaN(d.getTime()) ? null : d;
  },

  mw_ageStr: function (dob) {
    var d0 = this.mw_parseDate(dob);
    if (!d0) { return ''; }
    var days = Math.floor((new Date().getTime() - d0.getTime()) / 86400000);
    if (days < 0) { return 'future'; }
    return days + 'd (' + Math.floor(days / 7) + 'w)';
  },

  mw_selText: function (id) {
    var sel = document.getElementById(id);
    if (!sel || sel.selectedIndex < 0) { return ''; }
    return sel.options[sel.selectedIndex].text;
  },

  mw_msg: function (id, text, isError) {
    var el = document.getElementById(id);
    if (!el) { return; }
    el.textContent = text ? text : '';
    if (isError) { el.style.color = '#c0392b'; } else { el.style.color = '#2e8b57'; }
    if (text) {
      window.setTimeout(function () {
        var e2 = document.getElementById(id);
        if (e2) { e2.textContent = ''; }
      }, 3000);
    }
  },

  mw_findMouse: function (mid) {
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      if (this.mw_mice[i].id === mid) { return this.mw_mice[i]; }
    }
    return null;
  },

  // Shared image downscaler -> callback(dataUrl). Used by shared + per-mouse.
  mw_shrinkImage: function (file, cb) {
    var maxDim = 800;
    if (!file.type || file.type.indexOf('image/') !== 0) { return; }
    var reader = new FileReader();
    reader.onload = function (ev) {
      var img = new Image();
      img.onload = function () {
        var w = img.width, h = img.height;
        if (w > h) { if (w > maxDim) { h = Math.round(h * maxDim / w); w = maxDim; } }
        else { if (h > maxDim) { w = Math.round(w * maxDim / h); h = maxDim; } }
        var canvas = document.createElement('canvas');
        canvas.width = w; canvas.height = h;
        var ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0, w, h);
        var dataUrl;
        try { dataUrl = canvas.toDataURL('image/jpeg', 0.7); } catch (err) { dataUrl = ''; }
        if (dataUrl) { cb(dataUrl); }
      };
      img.onerror = function () {};
      img.src = ev.target.result;
    };
    reader.onerror = function () {};
    reader.readAsDataURL(file);
  },

  // ================= add / update / remove mouse =================
  mw_readMouseForm: function () {
    var m = {};
    m.id = this.mw_trim(document.getElementById('m_id').value);
    m.eartag = this.mw_trim(document.getElementById('m_eartag').value);
    m.sex = this.mw_selText('m_sex');
    m.cage = this.mw_trim(document.getElementById('m_cage').value);
    m.group = this.mw_trim(document.getElementById('m_group').value);
    m.dob = this.mw_trim(document.getElementById('m_dob').value);
    m.strain = this.mw_trim(document.getElementById('m_strain').value);
    m.genotype = this.mw_trim(document.getElementById('m_genotypeother').value);
    return m;
  },

  mw_addMouse: function () {
    var m = this.mw_readMouseForm();
    if (!m.id) { this.mw_msg('mousemsg', 'Mouse ID is required.', true); return; }

    if (this.mw_editId) {
      var oldId = this.mw_editId;
      if (m.id !== oldId && this.mw_findMouse(m.id)) {
        this.mw_msg('mousemsg', 'That Mouse ID already exists.', true); return;
      }
      var target = this.mw_findMouse(oldId);
      if (target) {
        target.id = m.id; target.eartag = m.eartag; target.sex = m.sex;
        target.cage = m.cage; target.group = m.group; target.dob = m.dob;
        target.strain = m.strain; target.genotype = m.genotype;
      }
      // rename inside events' mice lists AND per-mouse maps
      if (m.id !== oldId) {
        var k;
        for (k = 0; k < this.mw_events.length; k++) {
          var ev = this.mw_events[k];
          var n;
          for (n = 0; n < ev.mice.length; n++) { if (ev.mice[n] === oldId) { ev.mice[n] = m.id; } }
          if (ev.perMouse && ev.perMouse[oldId] !== undefined) { ev.perMouse[m.id] = ev.perMouse[oldId]; delete ev.perMouse[oldId]; }
          if (ev.perMousePhotos && ev.perMousePhotos[oldId] !== undefined) { ev.perMousePhotos[m.id] = ev.perMousePhotos[oldId]; delete ev.perMousePhotos[oldId]; }
        }
      }
      this.mw_endEdit();
      this.mw_renderRoster(); this.mw_renderMiceChecks(); this.mw_renderGroupControls();
      this.mw_renderFilter(); this.mw_renderLog();
      this.mw_msg('mousemsg', 'Mouse updated.', false);
      if (window.MLmarkDirty) { window.MLmarkDirty(); }
      return;
    }

    if (this.mw_findMouse(m.id)) { this.mw_msg('mousemsg', 'That Mouse ID already exists.', true); return; }
    this.mw_mice.push(m);
    document.getElementById('m_id').value = '';
    document.getElementById('m_eartag').value = '';
    this.mw_renderRoster(); this.mw_renderMiceChecks(); this.mw_renderGroupControls();
    this.mw_renderFilter(); this.mw_renderLog();
    this.mw_msg('mousemsg', 'Mouse added. Shared fields kept for next littermate.', false);
    if (window.MLmarkDirty) { window.MLmarkDirty(); }
  },

  mw_editMouse: function (mid) {
    var m = this.mw_findMouse(mid);
    if (!m) { return; }
    document.getElementById('m_id').value = m.id;
    document.getElementById('m_eartag').value = m.eartag;
    document.getElementById('m_sex').value = m.sex ? this.mw_valueForText('m_sex', m.sex) : '';
    document.getElementById('m_cage').value = m.cage;
    document.getElementById('m_group').value = m.group || '';
    document.getElementById('m_dob').value = m.dob;
    document.getElementById('m_strain').value = m.strain;
    document.getElementById('m_genotypeother').value = m.genotype || '';
    var gsel = document.getElementById('m_genotype');
    var matched = this.mw_valueForText('m_genotype', m.genotype);
    if (matched) { gsel.value = matched; }
    else if (m.genotype) { gsel.value = 'other'; }
    else { gsel.value = ''; }

    this.mw_editId = mid;
    var btn = document.getElementById('addmousebtn');
    if (btn) { btn.textContent = 'Update Mouse'; }
    var cbtn = document.getElementById('clearmousebtn');
    if (cbtn) { cbtn.textContent = 'Cancel Edit'; }
    this.mw_msg('mousemsg', 'Editing ' + mid + ' - make changes and click Update.', false);
    var idf = document.getElementById('m_id');
    if (idf && idf.scrollIntoView) { idf.scrollIntoView({ block: 'center' }); }
  },

  mw_valueForText: function (selId, str) {
    var sel = document.getElementById(selId);
    if (!sel) { return ''; }
    var i;
    for (i = 0; i < sel.options.length; i++) {
      if (sel.options[i].text === str || sel.options[i].value === str) { return sel.options[i].value; }
    }
    return '';
  },

  mw_endEdit: function () {
    this.mw_editId = null;
    var btn = document.getElementById('addmousebtn'); if (btn) { btn.textContent = 'Add Mouse'; }
    var cbtn = document.getElementById('clearmousebtn'); if (cbtn) { cbtn.textContent = 'Clear'; }
  },

  mw_clearMouse: function () {
    var ids = ['m_id','m_eartag','m_sex','m_cage','m_group','m_dob','m_strain','m_genotype','m_genotypeother'];
    var i;
    for (i = 0; i < ids.length; i++) { document.getElementById(ids[i]).value = ''; }
    this.mw_endEdit();
  },

  mw_removeMouse: function (mid) {
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      if (this.mw_mice[i].id === mid) { this.mw_mice.splice(i, 1); break; }
    }
    var keep = [];
    var k;
    for (k = 0; k < this.mw_events.length; k++) {
      var ev = this.mw_events[k];
      var m2 = [];
      var n;
      for (n = 0; n < ev.mice.length; n++) { if (ev.mice[n] !== mid) { m2.push(ev.mice[n]); } }
      ev.mice = m2;
      if (ev.perMouse && ev.perMouse[mid] !== undefined) { delete ev.perMouse[mid]; }
      if (ev.perMousePhotos && ev.perMousePhotos[mid] !== undefined) { delete ev.perMousePhotos[mid]; }
      if (ev.mice.length > 0) { keep.push(ev); }
    }
    this.mw_events = keep;
    this.mw_renderRoster(); this.mw_renderMiceChecks(); this.mw_renderGroupControls();
    this.mw_renderFilter(); this.mw_renderLog();
    if (window.MLmarkDirty) { window.MLmarkDirty(); }
  },

  mw_renderRoster: function () {
    var self = this;
    var body = document.getElementById('rosterbody');
    if (!body) { return; }
    if (this.mw_mice.length === 0) {
      body.innerHTML = '<' + 'tr><' + 'td colspan="10" style="text-align:center;color:#6b7885;font-style:italic;">No mice registered yet.<' + '/td><' + '/tr>';
      this.mw_applyRosterCollapse();
      return;
    }
    var html = '';
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      var m = this.mw_mice[i];
      html = html + '<' + 'tr>';
      html = html + '<' + 'td>' + this.mw_esc(m.id) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.eartag) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.sex) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.cage) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.group || '') + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.dob) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(this.mw_ageStr(m.dob)) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.genotype) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.strain) + '<' + '/td>';
      html = html + '<' + 'td style="white-space:nowrap;">' +
             '<' + 'button type="button" class="mw-btn sec mw-editmouse" data-id="' + this.mw_esc(m.id) + '" style="padding:2px 7px;margin-right:4px;">edit<' + '/button>' +
             '<' + 'button type="button" class="mw-btn sec mw-delmouse" data-id="' + this.mw_esc(m.id) + '" style="padding:2px 7px;">x<' + '/button>' +
             '<' + '/td>';
      html = html + '<' + '/tr>';
    }
    body.innerHTML = html;

    var edits = body.querySelectorAll('.mw-editmouse');
    var q;
    for (q = 0; q < edits.length; q++) {
      edits[q].addEventListener('click', function () { self.mw_editMouse(this.getAttribute('data-id')); });
    }
    var dels = body.querySelectorAll('.mw-delmouse');
    var j;
    for (j = 0; j < dels.length; j++) {
      dels[j].addEventListener('click', function () { self.mw_removeMouse(this.getAttribute('data-id')); });
    }
    this.mw_applyRosterCollapse();
  },

  // ================= groups =================
  mw_groupList: function () {
    var seen = {}; var out = []; var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      var g = this.mw_mice[i].group;
      if (g && !seen[g]) { seen[g] = true; out.push(g); }
    }
    out.sort();
    return out;
  },

  mw_renderGroupControls: function () {
    var sel = document.getElementById('e_groupsel');
    if (!sel) { return; }
    var prev = sel.value;
    var groups = this.mw_groupList();
    var html = '<' + 'option value="">-- add a group --<' + '/option>';
    var i;
    for (i = 0; i < groups.length; i++) {
      html = html + '<' + 'option value="' + this.mw_esc(groups[i]) + '">' + this.mw_esc(groups[i]) + '<' + '/option>';
    }
    sel.innerHTML = html;
    var found = false; var o;
    for (o = 0; o < sel.options.length; o++) { if (sel.options[o].value === prev) { found = true; break; } }
    sel.value = found ? prev : '';
  },

  mw_addGroupMice: function () {
    var sel = document.getElementById('e_groupsel');
    if (!sel || !sel.value) { this.mw_msg('eventmsg', 'Pick a group first.', true); return; }
    var g = sel.value;
    var chks = document.querySelectorAll('.mw-micechk');
    var count = 0; var i;
    for (i = 0; i < chks.length; i++) {
      var m = this.mw_findMouse(chks[i].value);
      if (m && m.group === g) { chks[i].checked = true; count++; }
    }
    this.mw_renderPerMouse();
    this.mw_msg('eventmsg', 'Ticked ' + count + ' mouse(s) in ' + g + '.', false);
  },

  mw_selectAllMice: function (on) {
    var chks = document.querySelectorAll('.mw-micechk');
    var i;
    for (i = 0; i < chks.length; i++) { chks[i].checked = on; }
    this.mw_renderPerMouse();
  },

  // ================= event mouse checkboxes =================
  mw_renderMiceChecks: function () {
    var box = document.getElementById('e_micechecks');
    if (!box) { return; }
    if (this.mw_mice.length === 0) {
      box.innerHTML = '<' + 'span style="color:#6b7885;font-style:italic;">Add mice above first.<' + '/span>';
      return;
    }
    var html = '';
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      var m = this.mw_mice[i];
      var lbl = this.mw_esc(m.id);
      if (m.eartag) { lbl = lbl + ' (' + this.mw_esc(m.eartag) + ')'; }
      if (m.group) { lbl = lbl + ' [' + this.mw_esc(m.group) + ']'; }
      html = html + '<' + 'label><' + 'input type="checkbox" class="mw-micechk" value="' + this.mw_esc(m.id) + '">' + lbl + '<' + '/label>';
    }
    box.innerHTML = html;

    var self = this;
    var chks = box.querySelectorAll('.mw-micechk');
    var c;
    for (c = 0; c < chks.length; c++) {
      chks[c].addEventListener('change', function () { self.mw_renderPerMouse(); });
    }
    this.mw_renderPerMouse();
  },

  mw_getCheckedMice: function () {
    var out = [];
    var chks = document.querySelectorAll('.mw-micechk');
    var i;
    for (i = 0; i < chks.length; i++) { if (chks[i].checked) { out.push(chks[i].value); } }
    return out;
  },

  // Render note input + photo picker + photo thumbnails for each checked mouse.
  mw_renderPerMouse: function () {
    var self = this;
    var wrap = document.getElementById('e_permousewrap');
    var box = document.getElementById('e_permouse');
    if (!wrap || !box) { return; }

    this.mw_capturePerMouse();

    var checked = this.mw_getCheckedMice();
    if (checked.length === 0) { box.innerHTML = ''; wrap.style.display = 'none'; return; }
    wrap.style.display = 'block';

    var html = '';
    var i;
    for (i = 0; i < checked.length; i++) {
      var mid = checked[i];
      var m = this.mw_findMouse(mid);
      var lbl = this.mw_esc(mid);
      if (m && m.eartag) { lbl = lbl + ' (' + this.mw_esc(m.eartag) + ')'; }
      var val = this.mw_stagePerMouse[mid] ? this.mw_esc(this.mw_stagePerMouse[mid]) : '';
      var eid = this.mw_esc(mid);

      html = html + '<' + 'div class="mw-pmrow">';
      html = html + '<' + 'span class="mw-pmlabel">' + lbl + '<' + '/span>';
      html = html + '<' + 'div class="mw-pmfields">';
      html = html + '<' + 'input type="text" class="mw-pminput" data-id="' + eid + '" value="' + val + '" placeholder="note specific to this mouse">';
      html = html + '<' + 'div style="margin-top:4px;"><' + 'label style="font-size:10px;color:#6b7885;font-weight:bold;">Photos for this mouse: <' + '/label>';
      html = html + '<' + 'input type="file" class="mw-pmphotoinput" data-id="' + eid + '" accept="image/*" multiple style="font-size:11px;"><' + '/div>';
      html = html + '<' + 'div class="mw-pmphotos" data-id="' + eid + '"><' + '/div>';
      html = html + '<' + '/div>';
      html = html + '<' + '/div>';
    }
    box.innerHTML = html;

    // wire photo inputs
    var inputs = box.querySelectorAll('.mw-pmphotoinput');
    var a;
    for (a = 0; a < inputs.length; a++) {
      inputs[a].addEventListener('change', function () {
        self.mw_handlePerMousePhotoInput(this.getAttribute('data-id'), this.files);
        this.value = '';
      });
    }
    // draw any staged per-mouse photo thumbnails
    for (i = 0; i < checked.length; i++) { this.mw_renderPerMousePhotos(checked[i]); }
  },

  mw_capturePerMouse: function () {
    var ins = document.querySelectorAll('.mw-pminput');
    var i;
    for (i = 0; i < ins.length; i++) {
      var id = ins[i].getAttribute('data-id');
      var v = this.mw_trim(ins[i].value);
      if (v) { this.mw_stagePerMouse[id] = v; }
      else { if (this.mw_stagePerMouse[id]) { delete this.mw_stagePerMouse[id]; } }
    }
  },

  mw_handlePerMousePhotoInput: function (mid, fileList) {
    var self = this;
    if (!fileList || fileList.length === 0) { return; }
    if (!this.mw_stagePerMousePhotos[mid]) { this.mw_stagePerMousePhotos[mid] = []; }
    var i;
    for (i = 0; i < fileList.length; i++) {
      this.mw_shrinkImage(fileList[i], function (dataUrl) {
        self.mw_stagePerMousePhotos[mid].push(dataUrl);
        self.mw_renderPerMousePhotos(mid);
      });
    }
  },

  mw_renderPerMousePhotos: function (mid) {
    var self = this;
    var box = document.querySelector('.mw-pmphotos[data-id="' + mid.replace(/"/g,'') + '"]');
    if (!box) { return; }
    var arr = this.mw_stagePerMousePhotos[mid] || [];
    if (arr.length === 0) { box.innerHTML = ''; return; }
    var html = '';
    var i;
    for (i = 0; i < arr.length; i++) {
      html = html + '<' + 'span class="mw-pmthumb">' +
             '<' + 'img src="' + arr[i] + '" data-mid="' + this.mw_esc(mid) + '" data-i="' + i + '" class="mw-pmthumbimg">' +
             '<' + 'button type="button" class="mw-pmthumbx" data-mid="' + this.mw_esc(mid) + '" data-i="' + i + '">x<' + '/button>' +
             '<' + '/span>';
    }
    box.innerHTML = html;

    var xs = box.querySelectorAll('.mw-pmthumbx');
    var a;
    for (a = 0; a < xs.length; a++) {
      xs[a].addEventListener('click', function () {
        var mm = this.getAttribute('data-mid');
        var ix = parseInt(this.getAttribute('data-i'), 10);
        if (self.mw_stagePerMousePhotos[mm]) { self.mw_stagePerMousePhotos[mm].splice(ix, 1); self.mw_renderPerMousePhotos(mm); }
      });
    }
    var ims = box.querySelectorAll('.mw-pmthumbimg');
    var b;
    for (b = 0; b < ims.length; b++) {
      ims[b].addEventListener('click', function () { self.mw_openPhoto(this.getAttribute('src')); });
    }
  },

  // ================= shared photos =================
  mw_handlePhotoInput: function (fileList) {
    var self = this;
    if (!fileList || fileList.length === 0) { return; }
    var i;
    for (i = 0; i < fileList.length; i++) {
      this.mw_shrinkImage(fileList[i], function (dataUrl) {
        self.mw_stagePhotos.push(dataUrl);
        self.mw_renderStagePhotos();
      });
    }
    var inp = document.getElementById('e_photos');
    if (inp) { inp.value = ''; }
  },

  mw_renderStagePhotos: function () {
    var self = this;
    var box = document.getElementById('e_photoprev');
    if (!box) { return; }
    if (this.mw_stagePhotos.length === 0) { box.innerHTML = ''; return; }
    var html = '';
    var i;
    for (i = 0; i < this.mw_stagePhotos.length; i++) {
      html = html + '<' + 'span class="mw-thumb">' +
             '<' + 'img src="' + this.mw_stagePhotos[i] + '" data-i="' + i + '" class="mw-stagethumb">' +
             '<' + 'button type="button" class="mw-thumbx mw-stagex" data-i="' + i + '">x<' + '/button>' +
             '<' + '/span>';
    }
    box.innerHTML = html;

    var xs = box.querySelectorAll('.mw-stagex');
    var a;
    for (a = 0; a < xs.length; a++) {
      xs[a].addEventListener('click', function () {
        self.mw_stagePhotos.splice(parseInt(this.getAttribute('data-i'), 10), 1);
        self.mw_renderStagePhotos();
      });
    }
    var ims = box.querySelectorAll('.mw-stagethumb');
    var b;
    for (b = 0; b < ims.length; b++) {
      ims[b].addEventListener('click', function () { self.mw_openPhoto(this.getAttribute('src')); });
    }
  },

  mw_openPhoto: function (dataUrl) {
    try {
      var w = window.open();

  mw_collectPerMouseData: function (mice) {
    // returns {perMouse:{}, perMousePhotos:{}} limited to the given mice
    this.mw_capturePerMouse();
    var perMouse = {};
    var perMousePhotos = {};
    var i;
    for (i = 0; i < mice.length; i++) {
      var mid = mice[i];
      if (this.mw_stagePerMouse[mid]) { perMouse[mid] = this.mw_stagePerMouse[mid]; }
      if (this.mw_stagePerMousePhotos[mid] && this.mw_stagePerMousePhotos[mid].length) {
        perMousePhotos[mid] = this.mw_stagePerMousePhotos[mid].slice(0);
      }
    }
    return { perMouse: perMouse, perMousePhotos: perMousePhotos };
  },

  // ================= CSV import / merge =================
  mw_detectDelim: function (line) { var tabs = line.split('\t').length - 1; var commas = line.split(',').length - 1; return (tabs >= commas && tabs > 0) ? '\t' : ','; },

  mw_parseLine: function (line, delim) {
    var out = []; var field = ''; var inQ = false; var i, c, next;
    for (i = 0; i < line.length; i++) {
      c = line.charAt(i);
      if (inQ) {
        if (c === '"') { next = line.charAt(i + 1); if (next === '"') { field = field + '"'; i++; } else { inQ = false; } }
        else { field = field + c; }
      } else {
        if (c === '"') { inQ = true; }
        else if (c === delim) { out.push(field); field = ''; }
        else { field = field + c; }
      }
    }
    out.push(field);
    return out;
  },

  mw_parseCsv: function (text) {
    text = String(text).replace(/\r\n/g, '\n').replace(/\r/g, '\n');
    var lines = text.split('\n');
    var rows = []; var delim = null; var i;
    for (i = 0; i < lines.length; i++) {
      if (this.mw_trim(lines[i]) === '') { continue; }
      if (delim === null) { delim = this.mw_detectDelim(lines[i]); }
      rows.push(this.mw_parseLine(lines[i], delim));
    }
    return rows;
  },

      if (!aliases.hasOwnProperty(key)) { continue; }
      map[key] = -1;
      for (j = 0; j < aliases[key].length && map[key] === -1; j++) {
        for (k = 0; k < norm.length; k++) { if (norm[k] === aliases[key][j]) { map[key] = k; break; } }
      }
    }
    return map;
  },

  mw_importCsv: function () {
    var raw = document.getElementById('importbox').value;
    if (!this.mw_trim(raw)) { this.mw_msg('importmsg', 'Paste CSV or Excel rows first.', true); return; }
    var dupMouse = this.mw_getRadio('dupmouse');
    var dupEvent = this.mw_getRadio('dupevent');
    var rows = this.mw_parseCsv(raw);
    if (rows.length < 1) { this.mw_msg('importmsg', 'No rows found.', true); return; }

    var map = this.mw_mapHeaders(rows[0]);
    var hasHeader = (map.id >= 0 || map.edate >= 0 || map.etype >= 0);
    var start = 0;
    if (hasHeader) { start = 1; }
    else { map = { id:0,eartag:1,sex:2,cage:3,group:4,dob:5,genotype:7,strain:8,edate:9,etype:10,user:11,fileloc:12,protocol:13,details:14,note:15 }; }
    if (map.id < 0) { this.mw_msg('importmsg', 'Could not find a "Mouse ID" column. Check the header row.', true); return; }

    var added = { mice: 0, events: 0 }, updated = { mice: 0, events: 0 }, skipped = { mice: 0, events: 0 }, badrows = 0;
    var r;
    for (r = start; r < rows.length; r++) {
      var col = rows[r];
      if (!col || col.length === 0) { continue; }
      var id = this.mw_cell(col, map.id);
      if (!id) { badrows++; continue; }

      var incoming = {
        id: id, eartag: this.mw_cell(col, map.eartag), sex: this.mw_cell(col, map.sex), cage: this.mw_cell(col, map.cage),
        group: this.mw_cell(col, map.group), dob: this.mw_cell(col, map.dob), strain: this.mw_cell(col, map.strain), genotype: this.mw_cell(col, map.genotype)
      };
      var existing = this.mw_findMouse(id);
      if (!existing) { this.mw_mice.push(incoming); added.mice++; existing = incoming; }
      else if (dupMouse === 'overwrite') {
        existing.eartag = incoming.eartag; existing.sex = incoming.sex; existing.cage = incoming.cage; existing.group = incoming.group;
        existing.dob = incoming.dob; existing.strain = incoming.strain; existing.genotype = incoming.genotype; updated.mice++;
      } else { skipped.mice++; }

      var edate = this.mw_cell(col, map.edate);
      var etype = this.mw_cell(col, map.etype);
      if (edate || etype) {
        var incNote = this.mw_cell(col, map.note);
        var incEv = {
          date: edate, type: etype, user: this.mw_cell(col, map.user), fileloc: this.mw_cell(col, map.fileloc),
          protocol: this.mw_cell(col, map.protocol), details: this.mw_cell(col, map.details),
          photos: [], perMouse: {}, perMousePhotos: {}, mice: [id]
        };
        if (incNote) { incEv.perMouse[id] = incNote; }
        var match = null;
        var e;
        for (e = 0; e < this.mw_events.length; e++) {
          var ex = this.mw_events[e];
          var belongs = false;
          var z;
          for (z = 0; z < ex.mice.length; z++) { if (ex.mice[z] === id) { belongs = true; break; } }
          if (belongs && this.mw_sameEvent(ex, incEv)) { match = ex; break; }
        }
        if (!match) { this.mw_events.push(incEv); added.events++; }
        else if (dupEvent === 'overwrite') {
          match.user = incEv.user; match.fileloc = incEv.fileloc; match.protocol = incEv.protocol; match.details = incEv.details;
          if (!match.perMouse) { match.perMouse = {}; }
          if (incNote) { match.perMouse[id] = incNote; }
          updated.events++;
        } else { skipped.events++; }
      }
    }

    this.mw_sortEvents();
    this.mw_renderAll();
    var summary = 'Mice +' + added.mice + ' / upd ' + updated.mice + ' / skip ' + skipped.mice + '  |  Events +' + added.events + ' / upd ' + updated.events + ' / skip ' + skipped.events;
    if (badrows > 0) { summary = summary + '  (' + badrows + ' row(s) had no Mouse ID)'; }
    this.mw_msg('importmsg', summary, false);
    if (window.MLmarkDirty) { window.MLmarkDirty(); }
  },

    if (!this.mw_events) { this.mw_events = []; }

      if (!this.mw_events[i].perMouse) { this.mw_events[i].perMouse = {}; }
      if (!this.mw_events[i].photos) { this.mw_events[i].photos = []; }
    }
    var tEl = document.getElementById('projecttitle');
    if (tEl) { tEl.value = title; }
  },

  // ----- CRC32 (needed by the ZIP format) -----
  mw_crc32: function (bytes) {
    var c;
    if (!this._crcTable) {
      this._crcTable = [];
      var n, k;
      for (n = 0; n < 256; n++) {
        c = n;
        for (k = 0; k < 8; k++) { c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1); }
        this._crcTable[n] = c >>> 0;
      }
    }
    var crc = 0xFFFFFFFF;
    var i;
    for (i = 0; i < bytes.length; i++) {
      crc = (crc >>> 8) ^ this._crcTable[(crc ^ bytes[i]) & 0xFF];
    }
    return (crc ^ 0xFFFFFFFF) >>> 0;
  },

  // ----- encoders -----
  mw_strToBytes: function (str) {
    // UTF-8 encode
    var utf8 = unescape(encodeURIComponent(str));
    var arr = new Uint8Array(utf8.length);
    var i;
    for (i = 0; i < utf8.length; i++) { arr[i] = utf8.charCodeAt(i) & 0xFF; }
    return arr;
  },

  mw_dataUrlToBytes: function (dataUrl) {
    var comma = dataUrl.indexOf(',');
    var b64 = dataUrl.substring(comma + 1);
    var bin = atob(b64);
    var n = bin.length;
    var arr = new Uint8Array(n);
    var i;
    for (i = 0; i < n; i++) { arr[i] = bin.charCodeAt(i); }
    return arr;
  },

  // ----- minimal STORE-only ZIP builder -----
  // files = [{ name: "path/in/zip", bytes: Uint8Array }]
  mw_buildZip: function (files) {
    var self = this;
    var chunks = [];       // array of Uint8Array to concatenate
    var central = [];      // central directory records
    var offset = 0;

    function u16(n) { return new Uint8Array([n & 0xFF, (n >>> 8) & 0xFF]); }
    function u32(n) { return new Uint8Array([n & 0xFF, (n >>> 8) & 0xFF, (n >>> 16) & 0xFF, (n >>> 24) & 0xFF]); }
    function push(arr) { chunks.push(arr); offset += arr.length; }
    function concat(list) {
      var total = 0, i;
      for (i = 0; i < list.length; i++) { total += list[i].length; }
      var out = new Uint8Array(total);
      var pos = 0;
      for (i = 0; i < list.length; i++) { out.set(list[i], pos); pos += list[i].length; }
      return out;
    }

    var i;
    for (i = 0; i < files.length; i++) {
      var nameBytes = self.mw_strToBytes(files[i].name);
      var data = files[i].bytes;
      var crc = self.mw_crc32(data);
      var localOffset = offset;

      // ---- local file header ----
      push(u32(0x04034b50));        // local file header signature
      push(u16(20));                // version needed
      push(u16(0));                 // flags
      push(u16(0));                 // compression = 0 (STORE)
      push(u16(0));                 // mod time
      push(u16(0));                 // mod date
      push(u32(crc));               // crc32
      push(u32(data.length));       // compressed size
      push(u32(data.length));       // uncompressed size
      push(u16(nameBytes.length));  // filename length
      push(u16(0));                 // extra length
      push(nameBytes);              // filename
      push(data);                   // file data

      // ---- central directory record (saved for later) ----
      var cd = [];
      cd.push(u32(0x02014b50));     // central dir signature
      cd.push(u16(20));             // version made by
      cd.push(u16(20));             // version needed
      cd.push(u16(0));              // flags
      cd.push(u16(0));              // compression
      cd.push(u16(0));              // mod time
      cd.push(u16(0));              // mod date
      cd.push(u32(crc));            // crc32
      cd.push(u32(data.length));    // compressed size
      cd.push(u32(data.length));    // uncompressed size
      cd.push(u16(nameBytes.length));
      cd.push(u16(0));              // extra length
      cd.push(u16(0));              // comment length
      cd.push(u16(0));              // disk number
      cd.push(u16(0));              // internal attrs
      cd.push(u32(0));              // external attrs
      cd.push(u32(localOffset));    // offset of local header
      cd.push(nameBytes);
      central.push({ bytes: concat(cd) });
    }

    // ---- write central directory ----
    var cdStart = offset;
    var cdBytes = [];
    for (i = 0; i < central.length; i++) { push(central[i].bytes); cdBytes.push(central[i].bytes); }
    var cdSize = offset - cdStart;

    // ---- end of central directory ----
    push(u32(0x06054b50));          // EOCD signature
    push(u16(0));                   // disk number
    push(u16(0));                   // disk with CD
    push(u16(files.length));        // entries on this disk
    push(u16(files.length));        // total entries
    push(u32(cdSize));              // CD size
    push(u32(cdStart));             // CD offset
    push(u16(0));                   // comment length

    return concat(chunks);
  },

  // ----- assemble the project's files and trigger the download -----
  mw_downloadZip: function () {
    var proj = this.mw_projectName();
    var safe = this.mw_zipSafeName(proj);
    var root = safe + '/';           // nested folder inside the zip
    var files = [];

    // 1) JSON (full state incl. photos) — the reload file
    files.push({ name: root + safe + '.json', bytes: this.mw_strToBytes(this.to_json()) });

    // 2) CSV (text copy)
    files.push({ name: root + safe + '.csv', bytes: this.mw_strToBytes(this.mw_buildCsv()) });

    // 3) photos/ — shared + per-mouse, same naming as the standalone
    var i;
    var photoCount = 0;
    for (i = 0; i < this.mw_events.length; i++) {
      var ev = this.mw_events[i];
      var base = safe + '_ev' + (i + 1);
      if (ev.photos && ev.photos.length) {
        var p;
        for (p = 0; p < ev.photos.length; p++) {
          files.push({ name: root + 'photos/' + base + '_shared_' + (p + 1) + '.jpg',
                       bytes: this.mw_dataUrlToBytes(ev.photos[p]) });
          photoCount++;
        }
      }
      if (ev.perMousePhotos) {
        var mid;
        for (mid in ev.perMousePhotos) {
          if (!ev.perMousePhotos.hasOwnProperty(mid)) { continue; }
          var arr = ev.perMousePhotos[mid] || [];
          var q;
          for (q = 0; q < arr.length; q++) {
            files.push({ name: root + 'photos/' + base + '_' + this.mw_zipSafeName(mid) + '_' + (q + 1) + '.jpg',
                         bytes: this.mw_dataUrlToBytes(arr[q]) });
            photoCount++;
          }
        }
      }
    }

    try {
      var zipBytes = this.mw_buildZip(files);
      var blob = new Blob([zipBytes], { type: 'application/zip' });
      var url = URL.createObjectURL(blob);
      var a = document.createElement('a');
      a.href = url;
      a.download = safe + '.zip';
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      URL.revokeObjectURL(url);
      this.mw_msg('exportmsg', 'ZIP downloaded (' + photoCount + ' photo file(s)).', false);
    } catch (err) {
      this.mw_msg('exportmsg', 'ZIP download failed here: ' + err.message, true);
    }
  }
