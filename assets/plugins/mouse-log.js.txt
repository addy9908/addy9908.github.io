my_widget_script =
{
  // ================= data =================
  mw_mice: [],    // [{id, eartag, sex, cage, dob, strain, genotype}]
  mw_events: [],  // [{date, type, protocol, user, fileloc, details, mice:[id,...]}]
  mw_editId: null, // id of mouse currently being edited, or null
  mw_editEventIdx: null, // index of event currently being edited, or null
  mw_stagePhotos: [], // data-URL strings staged for the event being composed/edited
  mw_stagePerMouse: {}, // { mouseId: "note" } staged per-mouse details
  mw_collapsed: {}, // { mouseId: true } collapsed groups in the log view
  mw_rosterCollapsed: false, // whether the roster table is collapsed

  // ================= helpers =================
  mw_esc: function (s) {
    if (s === null || s === undefined) { return ''; }
    s = String(s);
    s = s.split('&').join('&amp;');
    s = s.split('<').join('&lt;');
    s = s.split('>').join('&gt;');
    s = s.split('"').join('&quot;');
    return s;
  },

  mw_today: function () {
    // type="date" inputs require YYYY-MM-DD as their value
    var d = new Date();
    var mm = ('0' + (d.getMonth() + 1)).slice(-2);
    var dd = ('0' + d.getDate()).slice(-2);
    return d.getFullYear() + '-' + mm + '-' + dd;
  },

  mw_trim: function (s) {
    if (s === null || s === undefined) { return ''; }
    return String(s).replace(/^\s+/, '').replace(/\s+$/, '');
  },

  // Sort events by parsed date. Direction follows the Sort dropdown
  // (oldest first by default). Works for YYYY-MM-DD and MM/DD/YYYY.
  mw_sortEvents: function () {
    var self = this;
    var dir = 1;
    var ssel = document.getElementById('sortorder');
    if (ssel && ssel.value === 'newest') { dir = -1; }
    this.mw_events.sort(function (a, b) {
      var da = self.mw_parseDate(a.date);
      var db = self.mw_parseDate(b.date);
      var ta = da ? da.getTime() : 0;
      var tb = db ? db.getTime() : 0;
      if (ta < tb) { return -1 * dir; }
      if (ta > tb) { return 1 * dir; }
      return 0;
    });
  },

  // Age of a mouse (by DOB) on a specific event date. Blank if no/invalid DOB.
  mw_ageAt: function (dob, eventDate) {
    var d0 = this.mw_parseDate(dob);
    var d1 = this.mw_parseDate(eventDate);
    if (!d0 || !d1) { return ''; }
    var days = Math.floor((d1.getTime() - d0.getTime()) / 86400000);
    if (days < 0) { return ''; }
    return days + 'd (' + Math.floor(days / 7) + 'w)';
  },

  // parse MM/DD/YYYY (also tolerates YYYY-MM-DD from older data)
  mw_parseDate: function (s) {
    if (!s) { return null; }
    s = this.mw_trim(s);
    var m;
    // MM/DD/YYYY
    m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
    if (m) { return new Date(parseInt(m[3],10), parseInt(m[1],10)-1, parseInt(m[2],10)); }
    // YYYY-MM-DD (legacy)
    m = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
    if (m) { return new Date(parseInt(m[1],10), parseInt(m[2],10)-1, parseInt(m[3],10)); }
    var d = new Date(s);
    return isNaN(d.getTime()) ? null : d;
  },

  mw_ageStr: function (dob) {
    var d0 = this.mw_parseDate(dob);
    if (!d0) { return ''; }
    var days = Math.floor((new Date().getTime() - d0.getTime()) / 86400000);
    if (days < 0) { return 'future'; }
    return days + 'd (' + Math.floor(days / 7) + 'w)';
  },

  mw_selText: function (id) {
    var sel = document.getElementById(id);
    if (!sel || sel.selectedIndex < 0) { return ''; }
    return sel.options[sel.selectedIndex].text;
  },

  mw_msg: function (id, text, isError) {
    var el = document.getElementById(id);
    if (!el) { return; }
    el.textContent = text ? text : '';
    if (isError) { el.style.color = '#c0392b'; } else { el.style.color = '#2e8b57'; }
    if (text) {
      window.setTimeout(function () {
        var e2 = document.getElementById(id);
        if (e2) { e2.textContent = ''; }
      }, 3000);
    }
  },

  mw_findMouse: function (mid) {
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      if (this.mw_mice[i].id === mid) { return this.mw_mice[i]; }
    }
    return null;
  },

  // ================= add / update / remove mouse =================
  mw_readMouseForm: function () {
    var geno = this.mw_selText('m_genotype');
    var genoOther = this.mw_trim(document.getElementById('m_genotypeother').value);
    if (document.getElementById('m_genotype').value === 'other' && genoOther) { geno = genoOther; }

    var m = {};
    m.id = this.mw_trim(document.getElementById('m_id').value);
    m.eartag = this.mw_trim(document.getElementById('m_eartag').value);
    m.sex = this.mw_selText('m_sex');
    m.cage = this.mw_trim(document.getElementById('m_cage').value);
    m.group = this.mw_trim(document.getElementById('m_group').value);
    m.dob = this.mw_trim(document.getElementById('m_dob').value);
    m.strain = this.mw_trim(document.getElementById('m_strain').value);
    m.genotype = (geno === '-- select --') ? '' : geno;
    return m;
  },

  mw_addMouse: function () {
    var m = this.mw_readMouseForm();
    if (!m.id) { this.mw_msg('mousemsg', 'Mouse ID is required.', true); return; }

    if (this.mw_editId) {
      // ---- update existing mouse ----
      var oldId = this.mw_editId;
      // if ID changed, make sure the new one isn't a duplicate of a different mouse
      if (m.id !== oldId && this.mw_findMouse(m.id)) {
        this.mw_msg('mousemsg', 'That Mouse ID already exists.', true); return;
      }
      var target = this.mw_findMouse(oldId);
      if (target) {
        target.id = m.id;
        target.eartag = m.eartag;
        target.sex = m.sex;
        target.cage = m.cage;
        target.group = m.group;
        target.dob = m.dob;
        target.strain = m.strain;
        target.genotype = m.genotype;
      }
      // if the ID was renamed, update it inside every event's mice list
      if (m.id !== oldId) {
        var k;
        for (k = 0; k < this.mw_events.length; k++) {
          var arr = this.mw_events[k].mice;
          var n;
          for (n = 0; n < arr.length; n++) {
            if (arr[n] === oldId) { arr[n] = m.id; }
          }
        }
      }
      this.mw_endEdit();
      this.mw_renderRoster();
      this.mw_renderMiceChecks();
      this.mw_renderGroupControls();
      this.mw_renderFilter();
      this.mw_renderLog();
      this.mw_msg('mousemsg', 'Mouse updated.', false);
      return;
    }

    // ---- add new mouse ----
    if (this.mw_findMouse(m.id)) { this.mw_msg('mousemsg', 'That Mouse ID already exists.', true); return; }
    this.mw_mice.push(m);
    // NOTE: fields are intentionally NOT cleared, so shared values
    // (cage, DOB, strain, genotype) carry to the next littermate.
    // Only clear the two fields that must be unique per animal.
    document.getElementById('m_id').value = '';
    document.getElementById('m_eartag').value = '';
    this.mw_renderRoster();
    this.mw_renderMiceChecks();
    this.mw_renderGroupControls();
    this.mw_renderFilter();
    this.mw_renderLog();
    this.mw_msg('mousemsg', 'Mouse added. Shared fields kept for next littermate.', false);
  },

  mw_editMouse: function (mid) {
    var m = this.mw_findMouse(mid);
    if (!m) { return; }
    document.getElementById('m_id').value = m.id;
    document.getElementById('m_eartag').value = m.eartag;
    document.getElementById('m_sex').value = m.sex ? this.mw_valueForText('m_sex', m.sex) : '';
    document.getElementById('m_cage').value = m.cage;
    document.getElementById('m_group').value = m.group || '';
    document.getElementById('m_dob').value = m.dob;
    document.getElementById('m_strain').value = m.strain;

    // genotype: match to a preset value, else use Other + free text
    var gsel = document.getElementById('m_genotype');
    var matched = this.mw_valueForText('m_genotype', m.genotype);
    if (matched) {
      gsel.value = matched;
      document.getElementById('m_genotypeother').value = '';
    } else if (m.genotype) {
      gsel.value = 'other';
      document.getElementById('m_genotypeother').value = m.genotype;
    } else {
      gsel.value = '';
      document.getElementById('m_genotypeother').value = '';
    }

    this.mw_editId = mid;
    var btn = document.getElementById('addmousebtn');
    if (btn) { btn.textContent = 'Update Mouse'; }
    var cbtn = document.getElementById('clearmousebtn');
    if (cbtn) { cbtn.textContent = 'Cancel Edit'; }
    this.mw_msg('mousemsg', 'Editing ' + mid + ' - make changes and click Update.', false);
    // bring the form into view
    var idf = document.getElementById('m_id');
    if (idf && idf.scrollIntoView) { idf.scrollIntoView({ block: 'center' }); }
  },

  // find the option value whose text OR value matches a stored string
  mw_valueForText: function (selId, str) {
    var sel = document.getElementById(selId);
    if (!sel) { return ''; }
    var i;
    for (i = 0; i < sel.options.length; i++) {
      if (sel.options[i].text === str || sel.options[i].value === str) {
        return sel.options[i].value;
      }
    }
    return '';
  },

  mw_endEdit: function () {
    this.mw_editId = null;
    var btn = document.getElementById('addmousebtn');
    if (btn) { btn.textContent = 'Add Mouse'; }
    var cbtn = document.getElementById('clearmousebtn');
    if (cbtn) { cbtn.textContent = 'Clear'; }
  },

  mw_clearMouse: function () {
    document.getElementById('m_id').value = '';
    document.getElementById('m_eartag').value = '';
    document.getElementById('m_sex').value = '';
    document.getElementById('m_cage').value = '';
    document.getElementById('m_group').value = '';
    document.getElementById('m_dob').value = '';
    document.getElementById('m_strain').value = '';
    document.getElementById('m_genotype').value = '';
    document.getElementById('m_genotypeother').value = '';
    this.mw_endEdit();
  },

  mw_removeMouse: function (mid) {
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      if (this.mw_mice[i].id === mid) { this.mw_mice.splice(i, 1); break; }
    }
    // drop this mouse from any event; remove events left with no mice
    var keep = [];
    var k;
    for (k = 0; k < this.mw_events.length; k++) {
      var ev = this.mw_events[k];
      var m2 = [];
      var n;
      for (n = 0; n < ev.mice.length; n++) {
        if (ev.mice[n] !== mid) { m2.push(ev.mice[n]); }
      }
      ev.mice = m2;
      if (ev.mice.length > 0) { keep.push(ev); }
    }
    this.mw_events = keep;

    this.mw_renderRoster();
    this.mw_renderMiceChecks();
    this.mw_renderGroupControls();
    this.mw_renderFilter();
    this.mw_renderLog();
  },

  mw_renderRoster: function () {
    var self = this;
    var body = document.getElementById('rosterbody');
    if (!body) { return; }

    if (this.mw_mice.length === 0) {
      body.innerHTML = '<' + 'tr><' + 'td colspan="10" style="text-align:center;color:#6b7885;font-style:italic;">No mice registered yet.<' + '/td><' + '/tr>';
      this.mw_applyRosterCollapse();
      return;
    }
    var html = '';
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      var m = this.mw_mice[i];
      html = html + '<' + 'tr>';
      html = html + '<' + 'td>' + this.mw_esc(m.id) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.eartag) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.sex) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.cage) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.group || '') + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.dob) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(this.mw_ageStr(m.dob)) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.genotype) + '<' + '/td>';
      html = html + '<' + 'td>' + this.mw_esc(m.strain) + '<' + '/td>';
      html = html + '<' + 'td style="white-space:nowrap;">' +
             '<' + 'button type="button" class="mw-btn sec mw-editmouse" data-id="' + this.mw_esc(m.id) + '" style="padding:2px 7px;margin-right:4px;">edit<' + '/button>' +
             '<' + 'button type="button" class="mw-btn sec mw-delmouse" data-id="' + this.mw_esc(m.id) + '" style="padding:2px 7px;">x<' + '/button>' +
             '<' + '/td>';
      html = html + '<' + '/tr>';
    }
    body.innerHTML = html;

    var edits = body.querySelectorAll('.mw-editmouse');
    var q;
    for (q = 0; q < edits.length; q++) {
      edits[q].addEventListener('click', function () {
        self.mw_editMouse(this.getAttribute('data-id'));
      });
    }

    var dels = body.querySelectorAll('.mw-delmouse');
    var j;
    for (j = 0; j < dels.length; j++) {
      dels[j].addEventListener('click', function () {
        self.mw_removeMouse(this.getAttribute('data-id'));
      });
    }

    this.mw_applyRosterCollapse();
  },

  // ================= groups =================
  mw_groupList: function () {
    var seen = {};
    var out = [];
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      var g = this.mw_mice[i].group;
      if (g && !seen[g]) { seen[g] = true; out.push(g); }
    }
    out.sort();
    return out;
  },

  mw_renderGroupControls: function () {
    var sel = document.getElementById('e_groupsel');
    if (!sel) { return; }
    var prev = sel.value;
    var groups = this.mw_groupList();
    var html = '<' + 'option value="">-- add a group --<' + '/option>';
    var i;
    for (i = 0; i < groups.length; i++) {
      html = html + '<' + 'option value="' + this.mw_esc(groups[i]) + '">' + this.mw_esc(groups[i]) + '<' + '/option>';
    }
    sel.innerHTML = html;
    // restore previous selection if still valid
    var found = false;
    var o;
    for (o = 0; o < sel.options.length; o++) { if (sel.options[o].value === prev) { found = true; break; } }
    sel.value = found ? prev : '';
  },

  mw_addGroupMice: function () {
    var sel = document.getElementById('e_groupsel');
    if (!sel || !sel.value) { this.mw_msg('eventmsg', 'Pick a group first.', true); return; }
    var g = sel.value;
    var chks = document.querySelectorAll('.mw-micechk');
    var count = 0;
    var i;
    for (i = 0; i < chks.length; i++) {
      var m = this.mw_findMouse(chks[i].value);
      if (m && m.group === g) { chks[i].checked = true; count++; }
    }
    this.mw_renderPerMouse();
    this.mw_msg('eventmsg', 'Ticked ' + count + ' mouse(s) in ' + g + '.', false);
  },

  mw_selectAllMice: function (on) {
    var chks = document.querySelectorAll('.mw-micechk');
    var i;
    for (i = 0; i < chks.length; i++) { chks[i].checked = on; }
    this.mw_renderPerMouse();
  },

  // ================= event mouse checkboxes =================
  mw_renderMiceChecks: function () {
    var box = document.getElementById('e_micechecks');
    if (!box) { return; }
    if (this.mw_mice.length === 0) {
      box.innerHTML = '<' + 'span style="color:#6b7885;font-style:italic;">Add mice above first.<' + '/span>';
      return;
    }
    var html = '';
    var i;
    for (i = 0; i < this.mw_mice.length; i++) {
      var m = this.mw_mice[i];
      var lbl = this.mw_esc(m.id);
      if (m.eartag) { lbl = lbl + ' (' + this.mw_esc(m.eartag) + ')'; }
      if (m.group) { lbl = lbl + ' [' + this.mw_esc(m.group) + ']'; }
      html = html + '<' + 'label><' + 'input type="checkbox" class="mw-micechk" value="' + this.mw_esc(m.id) + '">' + lbl + '<' + '/label>';
    }
    box.innerHTML = html;

    // re-render per-mouse note inputs whenever the checked set changes
    var self = this;
    var chks = box.querySelectorAll('.mw-micechk');
    var c;
    for (c = 0; c < chks.length; c++) {
      chks[c].addEventListener('change', function () { self.mw_renderPerMouse(); });
    }
    this.mw_renderPerMouse();
  },

  mw_getCheckedMice: function () {
    var out = [];
    var chks = document.querySelectorAll('.mw-micechk');
    var i;
    for (i = 0; i < chks.length; i++) {
      if (chks[i].checked) { out.push(chks[i].value); }
    }
    return out;
  },

  // Render a note input for each currently-checked mouse, preserving
  // any text already typed (and any values in mw_stagePerMouse).
  mw_renderPerMouse: function () {
    var wrap = document.getElementById('e_permousewrap');
    var box = document.getElementById('e_permouse');
    if (!wrap || !box) { return; }

    // capture current typed values before rebuilding
    this.mw_capturePerMouse();

    var checked = this.mw_getCheckedMice();
    if (checked.length === 0) {
      box.innerHTML = '';
      wrap.style.display = 'none';
      return;
    }
    wrap.style.display = 'block';

    var html = '';
    var i;
    for (i = 0; i < checked.length; i++) {
      var mid = checked[i];
      var m = this.mw_findMouse(mid);
      var lbl = this.mw_esc(mid);
      if (m && m.eartag) { lbl = lbl + ' (' + this.mw_esc(m.eartag) + ')'; }
      var val = this.mw_stagePerMouse[mid] ? this.mw_esc(this.mw_stagePerMouse[mid]) : '';
      html = html + '<' + 'div class="mw-pmrow">' +
             '<' + 'span class="mw-pmlabel">' + lbl + '<' + '/span>' +
             '<' + 'input type="text" class="mw-pminput" data-id="' + this.mw_esc(mid) + '" value="' + val + '" placeholder="note specific to this mouse">' +
             '<' + '/div>';
    }
    box.innerHTML = html;
  },

  // Read current per-mouse inputs into mw_stagePerMouse (so values
  // survive re-render when the checked set changes).
  mw_capturePerMouse: function () {
    var ins = document.querySelectorAll('.mw-pminput');
    var i;
    for (i = 0; i < ins.length; i++) {
      var id = ins[i].getAttribute('data-id');
      var v = this.mw_trim(ins[i].value);
      if (v) { this.mw_stagePerMouse[id] = v; }
      else { if (this.mw_stagePerMouse[id]) { delete this.mw_stagePerMouse[id]; } }
    }
  },

  // ================= photos (downscaled thumbnails) =================
  // Read chosen files, shrink each to <= maxDim and JPEG-compress, then
  // push the resulting data URL into mw_stagePhotos and re-render preview.
  mw_handlePhotoInput: function (fileList) {
    var self = this;
    if (!fileList || fileList.length === 0) { return; }
    var maxDim = 800;
    var i;
    for (i = 0; i < fileList.length; i++) {
      (function (file) {
        if (!file.type || file.type.indexOf('image/') !== 0) { return; }
        var reader = new FileReader();
        reader.onload = function (ev) {
          var img = new Image();
          img.onload = function () {
            var w = img.width, h = img.height;
            if (w > h) {
              if (w > maxDim) { h = Math.round(h * maxDim / w); w = maxDim; }
            } else {
              if (h > maxDim) { w = Math.round(w * maxDim / h); h = maxDim; }
            }
            var canvas = document.createElement('canvas');
            canvas.width = w; canvas.height = h;
            var ctx = canvas.getContext('2d');
            ctx.drawImage(img, 0, 0, w, h);
            var dataUrl;
            try { dataUrl = canvas.toDataURL('image/jpeg', 0.7); }
            catch (err) { dataUrl = ''; }
            if (dataUrl) {
              self.mw_stagePhotos.push(dataUrl);
              self.mw_renderStagePhotos();
            }
          };
          img.onerror = function () {
            self.mw_msg('eventmsg', 'Could not read one image.', true);
          };
          img.src = ev.target.result;
        };
        reader.onerror = function () {
          self.mw_msg('eventmsg', 'File read blocked in this view.', true);
        };
        reader.readAsDataURL(file);
      })(fileList[i]);
    }
    // reset the input so the same file can be re-picked if needed
    var inp = document.getElementById('e_photos');
    if (inp) { inp.value = ''; }
  },

  mw_renderStagePhotos: function () {
    var self = this;
    var box = document.getElementById('e_photoprev');
    if (!box) { return; }
    if (this.mw_stagePhotos.length === 0) { box.innerHTML = ''; return; }
    var html = '';
    var i;
    for (i = 0; i < this.mw_stagePhotos.length; i++) {
      html = html + '<' + 'span class="mw-thumb">' +
             '<' + 'img src="' + this.mw_stagePhotos[i] + '" data-i="' + i + '" class="mw-stagethumb">' +
             '<' + 'button type="button" class="mw-thumbx mw-stagex" data-i="' + i + '">x<' + '/button>' +
             '<' + '/span>';
    }
    box.innerHTML = html;

    var xs = box.querySelectorAll('.mw-stagex');
    var a;
    for (a = 0; a < xs.length; a++) {
      xs[a].addEventListener('click', function () {
        self.mw_stagePhotos.splice(parseInt(this.getAttribute('data-i'), 10), 1);
        self.mw_renderStagePhotos();
      });
    }
    var ims = box.querySelectorAll('.mw-stagethumb');
    var b;
    for (b = 0; b < ims.length; b++) {
      ims[b].addEventListener('click', function () {
        self.mw_openPhoto(this.getAttribute('src'));
      });
    }
  },

  // Open a photo in a new tab/window. Sandbox may block window.open; if
  // so, the thumbnail itself is still visible inline.
  mw_openPhoto: function (dataUrl) {
    try {
      var w = window.open();
      if (w && w.document) {
        w.document.write('<' + 'img src="' + dataUrl + '" style="max-width:100%">');
      }
    } catch (err) {
      // silently ignore; inline thumbnail remains available
    }
  },

  // ================= add / update / remove event =================
  mw_addEvent: function () {
    var date = document.getElementById('e_date').value;
    if (!date) { this.mw_msg('eventmsg', 'Event date is required.', true); return; }
    var mice = this.mw_getCheckedMice();
    if (mice.length === 0) { this.mw_msg('eventmsg', 'Select at least one mouse.', true); return; }
    this.mw_capturePerMouse();

    // build a per-mouse note map limited to the mice on this event
    var perMouse = {};
    var pmI;
    for (pmI = 0; pmI < mice.length; pmI++) {
      if (this.mw_stagePerMouse[mice[pmI]]) { perMouse[mice[pmI]] = this.mw_stagePerMouse[mice[pmI]]; }
    }

    if (this.mw_editEventIdx !== null && this.mw_editEventIdx >= 0 &&
        this.mw_editEventIdx < this.mw_events.length) {
      // ---- update existing event ----
      var ev = this.mw_events[this.mw_editEventIdx];
      ev.date = this.mw_trim(date);
      ev.type = this.mw_selText('e_type');
      ev.protocol = this.mw_trim(document.getElementById('e_protocol').value);

  mw_rowHtml: function (ev, idx, showMice, forMouseId) {
    var details = this.mw_esc(ev.details);
    var protocol = this.mw_esc(ev.protocol);
    if (!protocol) { protocol = '&mdash;'; }

    // start with the common detail
    var dcell = details ? details : '';

    // append this mouse's specific note, if any
    if (forMouseId && ev.perMouse && ev.perMouse[forMouseId]) {
      var note = this.mw_esc(ev.perMouse[forMouseId]);
      if (dcell) {
        dcell = dcell + '<' + 'br><' + 'span style="color:#1f5578;">&raquo; ' + note + '<' + '/span>';
      } else {
        dcell = '<' + 'span style="color:#1f5578;">&raquo; ' + note + '<' + '/span>';
      }
    }
    if (!dcell) { dcell = '&mdash;'; }
    // build the meta line: age-at-event, experimenter, file location
    var meta = '';
    if (forMouseId) {
      var fm = this.mw_findMouse(forMouseId);
      if (fm && fm.dob) {
        var ageAt = this.mw_ageAt(fm.dob, ev.date);
        if (ageAt) { meta = 'Age: ' + ageAt; }
      }
    }
    if (ev.user) { if (meta) { meta = meta + ' | '; } meta = meta + 'By: ' + this.mw_esc(ev.user); }
    if (ev.fileloc) {
      if (meta) { meta = meta + ' | '; }
      meta = meta + 'File: ' + this.mw_esc(ev.fileloc);
    }
    if (meta) {
      dcell = dcell + '<' + 'br><' + 'span style="color:#6b7885;font-size:10px;">' + meta + '<' + '/span>';
    }

    // photo thumbnails (clickable)
    if (ev.photos && ev.photos.length) {
      dcell = dcell + '<' + 'div class="mw-logthumb">';
      var p;
      for (p = 0; p < ev.photos.length; p++) {
        dcell = dcell + '<' + 'img src="' + ev.photos[p] + '" class="mw-openphoto">';
      }
      dcell = dcell + '<' + '/div>';
    }

    var typeCell = this.mw_esc(ev.type);
    if (showMice) {
      typeCell = typeCell + '<' + 'br><' + 'span style="color:#6b7885;font-size:10px;">' + this.mw_esc(ev.mice.join(', ')) + '<' + '/span>';
    }
    var h = '<' + 'tr>';
    h = h + '<' + 'td>' + this.mw_esc(ev.date) + '<' + '/td>';
    h = h + '<' + 'td>' + typeCell + '<' + '/td>';
    h = h + '<' + 'td>' + dcell + '<' + '/td>';
    h = h + '<' + 'td>' + protocol + '<' + '/td>';
    h = h + '<' + 'td style="white-space:nowrap;">' +
        '<' + 'button type="button" class="mw-btn sec mw-editevent" data-i="' + idx + '" style="padding:2px 7px;margin-right:4px;">edit<' + '/button>' +
        '<' + 'button type="button" class="mw-btn sec mw-dupevent" data-i="' + idx + '" style="padding:2px 7px;margin-right:4px;">copy<' + '/button>' +
        '<' + 'button type="button" class="mw-btn sec mw-delevent" data-i="' + idx + '" style="padding:2px 7px;">x<' + '/button>' +
        '<' + '/td>';
    h = h + '<' + '/tr>';
    return h;
  },

  // render grouped-by-mouse blocks for a given list of mice
  mw_renderGroupedFor: function (miceArr) {
    var html = '';
    var i;
    for (i = 0; i < miceArr.length; i++) {
      var m = miceArr[i];
      var rows = this.mw_eventsForMouse(m.id);
      var isCollapsed = this.mw_collapsed[m.id] ? true : false;
      var arrow = isCollapsed ? '&#9656;' : '&#9662;'; // right triangle / down triangle
      var hdr = arrow + ' ' + this.mw_esc(m.id);
      if (m.eartag) { hdr = hdr + ' - tag ' + this.mw_esc(m.eartag); }
      if (m.group) { hdr = hdr + ' [' + this.mw_esc(m.group) + ']'; }
      hdr = hdr + ' (' + rows.length + ' event' + (rows.length === 1 ? '' : 's') + ')';
      html = html + '<' + 'tr class="mw-grouphdr mw-toggle" data-id="' + this.mw_esc(m.id) + '" style="cursor:pointer;"><' + 'td colspan="5">' + hdr + '<' + '/td><' + '/tr>';
      if (!isCollapsed) {
        if (rows.length === 0) {
          html = html + '<' + 'tr><' + 'td colspan="5" style="color:#6b7885;font-style:italic;">no events<' + '/td><' + '/tr>';
        } else {
          var r;
          for (r = 0; r < rows.length; r++) {
            html = html + this.mw_rowHtml(rows[r].ev, rows[r].idx, false, m.id);
          }
        }
      }
    }
    return html;
  },

  mw_renderLog: function () {
    var self = this;
    var body = document.getElementById('logbody');
    if (!body) { return; }

    var filter = '__all__';
    var fsel = document.getElementById('filtermouse');
    if (fsel) { filter = fsel.value; }

    if (this.mw_events.length === 0) {
      body.innerHTML = '<' + 'tr><' + 'td colspan="5" style="text-align:center;color:#6b7885;font-style:italic;">No events logged yet.<' + '/td><' + '/tr>';
      this.mw_wireLogDeletes();
      return;
    }

    var html = '';

    if (filter === '__all__') {
      // all mice, grouped by mouse
      html = this.mw_renderGroupedFor(this.mw_mice);

    } else if (filter.indexOf('__grp__') === 0) {
      // one group: show its mice, each grouped by mouse
      var gname = filter.substring(7);
      var inGroup = [];
      var a;
      for (a = 0; a < this.mw_mice.length; a++) {
        if (this.mw_mice[a].group === gname) { inGroup.push(this.mw_mice[a]); }
      }
      if (inGroup.length === 0) {
        html = '<' + 'tr><' + 'td colspan="5" style="color:#6b7885;font-style:italic;">No mice in group ' + this.mw_esc(gname) + '.<' + '/td><' + '/tr>';
      } else {
        html = this.mw_renderGroupedFor(inGroup);
      }

    } else {
      // single mouse
      var single = this.mw_eventsForMouse(filter);
      if (single.length === 0) {
        html = '<' + 'tr><' + 'td colspan="5" style="color:#6b7885;font-style:italic;">No events for ' + this.mw_esc(filter) + '.<' + '/td><' + '/tr>';
      } else {
        var s;
        for (s = 0; s < single.length; s++) {
          html = html + this.mw_rowHtml(single[s].ev, single[s].idx, false, filter);
        }
      }
    }

    body.innerHTML = html;
    this.mw_wireLogDeletes();
  },

      for (b = 0; b < rows[a].length; b++) { line.push(this.mw_csvCell(rows[a][b])); }

  // ================= CSV import / merge =================
  // Detect the delimiter of a line: tab (from Excel/Sheets copy) or comma.
  mw_detectDelim: function (line) {
    var tabs = line.split('\t').length - 1;
    var commas = line.split(',').length - 1;
    return (tabs >= commas && tabs > 0) ? '\t' : ',';
  },

  // Parse one delimited line, honoring quoted fields and "" escapes.
  mw_parseLine: function (line, delim) {
    var out = [];
    var field = '';
    var inQ = false;
    var i, c, next;
    for (i = 0; i < line.length; i++) {
      c = line.charAt(i);
      if (inQ) {
        if (c === '"') {
          next = line.charAt(i + 1);
          if (next === '"') { field = field + '"'; i++; }
          else { inQ = false; }
        } else { field = field + c; }
      } else {
        if (c === '"') { inQ = true; }
        else if (c === delim) { out.push(field); field = ''; }
        else { field = field + c; }
      }
    }
    out.push(field);
    return out;
  },

  // Parse full text into array of row-arrays. Delimiter detected from the
  // header (first non-empty) line, so tab- and comma-separated both work.
  mw_parseCsv: function (text) {
    text = String(text).replace(/\r\n/g, '\n').replace(/\r/g, '\n');
    var lines = text.split('\n');
    var rows = [];
    var delim = null;
    var i;
    for (i = 0; i < lines.length; i++) {
      if (this.mw_trim(lines[i]) === '') { continue; }
      if (delim === null) { delim = this.mw_detectDelim(lines[i]); }
      rows.push(this.mw_parseLine(lines[i], delim));
    }
    return rows;
  },

  mw_getRadio: function (name) {

    return '';
  },

  mw_sameEvent: function (a, b) {

  // Normalize a header cell for matching: lowercase, strip non-alphanumerics.
  mw_normHdr: function (s) {
    return String(s || '').toLowerCase().replace(/[^a-z0-9]/g, '');
  },

  // Build a name->index map from the header row using known aliases.
  mw_mapHeaders: function (headerRow) {
    var aliases = {
      id: ['mouseid', 'id', 'mouse'],
      eartag: ['eartag', 'tag'],
      sex: ['sex'],
      cage: ['cage', 'cagenumber'],
      group: ['group', 'batch', 'groupbatch'],
      dob: ['dob', 'dateofbirth', 'birth'],
      genotype: ['genotype'],
      strain: ['strain', 'background', 'strainbackground'],
      edate: ['eventdate', 'date'],
      etype: ['eventtype', 'type', 'event'],
      user: ['experimenter', 'user', 'operator'],
      fileloc: ['filelocation', 'fileloc', 'filestoragelocation', 'file'],
      protocol: ['protocol', 'iacuc', 'protocoliacuc'],
      details: ['details', 'commondetails', 'notes'],
      note: ['mousenote', 'note', 'permousenote', 'permousedetails']
    };
    var map = {};
    var norm = [];
    var i;
    for (i = 0; i < headerRow.length; i++) { norm.push(this.mw_normHdr(headerRow[i])); }

    var key, j, k;
    for (key in aliases) {
      if (!aliases.hasOwnProperty(key)) { continue; }
      map[key] = -1;
      for (j = 0; j < aliases[key].length && map[key] === -1; j++) {
        for (k = 0; k < norm.length; k++) {
          if (norm[k] === aliases[key][j]) { map[key] = k; break; }
        }
      }
    }
    return map;
  },

  // Safe cell fetch by mapped index.
  mw_cell: function (row, idx) {
    if (idx < 0 || idx >= row.length) { return ''; }
    return this.mw_trim(row[idx] || '');
  },

  mw_importCsv: function () {
    var raw = document.getElementById('importbox').value;
    if (!this.mw_trim(raw)) { this.mw_msg('importmsg', 'Paste CSV or Excel rows first.', true); return; }

    var dupMouse = this.mw_getRadio('dupmouse');   // skip | overwrite
    var dupEvent = this.mw_getRadio('dupevent');   // keep | overwrite

    var rows = this.mw_parseCsv(raw);
    if (rows.length < 1) { this.mw_msg('importmsg', 'No rows found.', true); return; }

    // Decide whether row 0 is a header. If it contains a recognizable
    // "mouse id" style column, treat it as a header and map by name.
    var map = this.mw_mapHeaders(rows[0]);
    var hasHeader = (map.id >= 0 || map.edate >= 0 || map.etype >= 0);
    var start = 0;

    if (hasHeader) {
      start = 1;
    } else {
      // No header: fall back to the known export column order.
      map = { id:0, eartag:1, sex:2, cage:3, group:4, dob:5, genotype:7, strain:8,
              edate:9, etype:10, user:11, fileloc:12, protocol:13, details:14, note:15 };
    }

    if (map.id < 0) {
      this.mw_msg('importmsg', 'Could not find a "Mouse ID" column. Check the header row.', true);
      return;
    }

    var added = { mice: 0, events: 0 };
    var updated = { mice: 0, events: 0 };
    var skipped = { mice: 0, events: 0 };
    var badrows = 0;

    var r;
    for (r = start; r < rows.length; r++) {
      var col = rows[r];
      if (!col || col.length === 0) { continue; }
      var id = this.mw_cell(col, map.id);
      if (!id) { badrows++; continue; }

      // ---- mouse ----
      var incoming = {
        id: id,
        eartag: this.mw_cell(col, map.eartag),
        sex: this.mw_cell(col, map.sex),
        cage: this.mw_cell(col, map.cage),
        group: this.mw_cell(col, map.group),
        dob: this.mw_cell(col, map.dob),
        strain: this.mw_cell(col, map.strain),
        genotype: this.mw_cell(col, map.genotype)
      };
      var existing = this.mw_findMouse(id);
      if (!existing) {
        this.mw_mice.push(incoming);
        added.mice++;
        existing = incoming;
      } else if (dupMouse === 'overwrite') {
        existing.eartag = incoming.eartag;
        existing.sex = incoming.sex;
        existing.cage = incoming.cage;
        existing.group = incoming.group;
        existing.dob = incoming.dob;
        existing.strain = incoming.strain;
        existing.genotype = incoming.genotype;
        updated.mice++;
      } else {
        skipped.mice++;
      }

      // ---- event (only if the row carries event data) ----
      var edate = this.mw_cell(col, map.edate);
      var etype = this.mw_cell(col, map.etype);
      if (edate || etype) {
        var incNote = this.mw_cell(col, map.note);
        var incEv = {
          date: edate,
          type: etype,
          user: this.mw_cell(col, map.user),
          fileloc: this.mw_cell(col, map.fileloc),
          protocol: this.mw_cell(col, map.protocol),
          details: this.mw_cell(col, map.details),
          photos: [],
          perMouse: {},
          mice: [id]
        };
        if (incNote) { incEv.perMouse[id] = incNote; }

        var match = null;
        var e;
        for (e = 0; e < this.mw_events.length; e++) {
          var ex = this.mw_events[e];
          var belongs = false;
          var z;
          for (z = 0; z < ex.mice.length; z++) { if (ex.mice[z] === id) { belongs = true; break; } }
          if (belongs && this.mw_sameEvent(ex, incEv)) { match = ex; break; }
        }
        if (!match) {
          this.mw_events.push(incEv);
          added.events++;
        } else if (dupEvent === 'overwrite') {
          match.user = incEv.user;
          match.fileloc = incEv.fileloc;
          match.protocol = incEv.protocol;
          match.details = incEv.details;
          if (!match.perMouse) { match.perMouse = {}; }
          if (incNote) { match.perMouse[id] = incNote; }
          updated.events++;
        } else {
          skipped.events++;
        }
      }
    }

    this.mw_renderAll();
    var summary = 'Mice +' + added.mice + ' / upd ' + updated.mice + ' / skip ' + skipped.mice +
                  '  |  Events +' + added.events + ' / upd ' + updated.events + ' / skip ' + skipped.events;
    if (badrows > 0) { summary = summary + '  (' + badrows + ' row(s) had no Mouse ID)'; }
    this.mw_msg('importmsg', summary, false);
  },

  mw_clearImport: function () {
    document.getElementById('importbox').value = '';
    this.mw_msg('importmsg', '', false);
  },

  // ================= wiring =================
  mw_wire: function () {
    var self = this;
    function bind(id, fn) {
      var el = document.getElementById(id);
      if (el && !el.getAttribute('data-mw')) {
        el.setAttribute('data-mw', '1');
        el.addEventListener('click', fn);
      }
    }
    bind('addmousebtn', function () { self.mw_addMouse(); });
    bind('clearmousebtn', function () { self.mw_clearMouse(); });
    bind('addeventbtn', function () { self.mw_addEvent(); });
    bind('cleareventbtn', function () { self.mw_clearEvent(); });
    bind('exportbtn', function () { self.mw_export(); });
    bind('copyexportbtn', function () { self.mw_copyExport(); });
    bind('downloadexportbtn', function () { self.mw_downloadExport(); });
    bind('importbtn', function () { self.mw_importCsv(); });
    bind('clearimportbtn', function () { self.mw_clearImport(); });
    bind('e_addgroupbtn', function () { self.mw_addGroupMice(); });
    bind('e_selallbtn', function () { self.mw_selectAllMice(true); });
    bind('e_clrallbtn', function () { self.mw_selectAllMice(false); });
    bind('collapseallbtn', function () { self.mw_collapseAll(true); });
    bind('expandallbtn', function () { self.mw_collapseAll(false); });
    bind('rosterhdr', function () { self.mw_toggleRoster(); });

    var fsel = document.getElementById('filtermouse');
    if (fsel && !fsel.getAttribute('data-mw')) {
      fsel.setAttribute('data-mw', '1');
      fsel.addEventListener('change', function () { self.mw_renderLog(); });
    }

    var ssel = document.getElementById('sortorder');
    if (ssel && !ssel.getAttribute('data-mw')) {
      ssel.setAttribute('data-mw', '1');
      ssel.addEventListener('change', function () { self.mw_sortEvents(); self.mw_renderLog(); });
    }

    var ed = document.getElementById('e_date');
    if (ed && !ed.value) { ed.value = this.mw_today(); }

    var pin = document.getElementById('e_photos');
    if (pin && !pin.getAttribute('data-mw')) {
      pin.setAttribute('data-mw', '1');
      pin.addEventListener('change', function () { self.mw_handlePhotoInput(this.files); });
    }
  },

  mw_renderAll: function () {
    this.mw_renderRoster();
    this.mw_renderMiceChecks();
    this.mw_renderGroupControls();
    this.mw_renderFilter();
    this.mw_renderLog();
  },

  mw_load: function (json_data) {
    this.mw_mice = [];
    this.mw_events = [];
    var title = '';
    try {
      var d = json_data;
      if (typeof d === 'string' && d.length) { d = JSON.parse(d); }
      if (d && typeof d === 'object') {
        if (d.mw_mice) { this.mw_mice = d.mw_mice; }
        if (d.mw_events) { this.mw_events = d.mw_events; }
        if (d.mw_title) { title = d.mw_title; }
      }
    } catch (err) {
      this.mw_mice = [];
      this.mw_events = [];
    }
    if (!this.mw_mice) { this.mw_mice = []; }
    if (!this.mw_events) { this.mw_events = []; }
    var tEl = document.getElementById('projecttitle');
    if (tEl) { tEl.value = title; }
  },

  // ================= default methods =================
  init: function (mode, json_data) {
    this.parent_class.init(mode, json_data, this);
    this.mw_load(json_data);
    this.mw_wire();
    this.mw_renderAll();
  },

  to_json: function () {
    var obj = {};
    var tEl = document.getElementById('projecttitle');
    obj.mw_title = tEl ? this.mw_trim(tEl.value) : '';
    obj.mw_mice = this.mw_mice;
    obj.mw_events = this.mw_events;
    return JSON.stringify(obj);
  },

  from_json: function (json_data) {
    this.parent_class.from_json(json_data);
    this.mw_load(json_data);
    this.mw_wire();
    this.mw_renderAll();
  },

  test_data: function () {
    return JSON.stringify({ mw_mice: [], mw_events: [] });
  },

  is_valid: function (b_suppress_message) {
    return this.parent_class.is_valid(b_suppress_message);
  },

  is_edited: function () {
    return this.parent_class.is_edited();
  },

  reset_edited: function () {
    return this.parent_class.reset_edited();
  }
